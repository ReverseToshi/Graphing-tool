cmake_minimum_required(VERSION 3.20)
project(MyWxApp)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

# Download wxWidgets
FetchContent_Declare(
    wxWidgets
    GIT_REPOSITORY https://github.com/wxWidgets/wxWidgets.git
    GIT_TAG        v3.2.4
)

# Let wxWidgets build only what we need (optional but faster)
set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(wxBUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(wxBUILD_SHARED ON CACHE BOOL "" FORCE)
set(wxUSE_STL OFF CACHE BOOL "" FORCE)

# Enable the minimal set of wx libraries
set(wxBUILD_MONOLITHIC OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(wxWidgets)

include_directories(${CMAKE_SOURCE_DIR}/headers)

# Add your application
add_executable(MyWxApp src/Main.cpp
                        src/App.cpp
                        src/MainFrame.cpp)

# Link against the main wxWidgets target directly
target_link_libraries(MyWxApp PRIVATE wx::core wx::base)

# Add include directories if needed
target_include_directories(MyWxApp PRIVATE ${wxWidgets_SOURCE_DIR}/include ${wxWidgets_BINARY_DIR}/lib/vc_x64_lib/mswu ${CMAKE_SOURCE_DIR}/headers)

# Define needed preprocessor macros (optional but recommended)
target_compile_definitions(MyWxApp PRIVATE WXUSINGDLL __WXMSW__)
